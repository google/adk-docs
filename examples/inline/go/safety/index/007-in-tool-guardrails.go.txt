// Helper function to check if a is a subset of b
func isSubset(a, b []string) bool {
	set := make(map[string]bool)
	for _, item := range b {
		set[item] = true
	}
	for _, item := range a {
		if _, found := set[item]; !found {
			return false
		}
	}
	return true
}
